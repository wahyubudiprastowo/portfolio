   192→  function walkKids(n, me) {
   193→    var kids = Array.prototype.slice.call(n.children).filter(function (k) { return k.tagName !== 'BR'; }), i = 0;

   197→      if (j - i >= 2) group(kids.slice(i, j), me); else walk(kids[i], me);
   198→      i = j;
   199→    }

   452→  /* ---- 派工：每個閒著的游標挑「離自己最近、可以開工」的圖層 ---- */
   453→  function pickTask(a) {
   454→    var best = null, bd = 1e9, seen = 0;
   455→    for (var i = 0; i < S.tasks.length && seen < 7; i++) {
   456→      var t = S.tasks[i];
   457→      if (t.taken || t.done) continue;
   458→      var ok = true;
   459→      for (var j = 0; j < t.deps.length; j++) if (!t.deps[j].done) { ok = false; break; }
   460→      if (!ok) continue;
   461→      seen++;
   462→      var d = Math.hypot(t.r.left - a.x, t.r.top - a.y) + i * 18;
   463→      if (d < bd) { bd = d; best = t; }
   464→    }
   465→    return best;
   466→  }

   599→    inView: function (r, pad) { pad = pad || 0; return r.width > 8 && r.height > 6 && r.top > 72 + pad && r.bottom < innerHeight - 24 - pad && r.left > -2 && r.right < innerWidth + 2; },
   600→    lines: function (e) {
   601→      var rg = document.createRange(); rg.selectNodeContents(e);
   602→      var rs = Array.prototype.slice.call(rg.getClientRects()), lines = [];

   608→        }

   612→      return lines.filter(function (L) { return L.right - L.left > 60 && self.inView({ width: L.right - L.left, height: L.bottom - L.top, top: L.top, bottom: L.bottom, left: L.left, right: L.right }); })
   613→                  .sort(function (p, q) { return p.top - q.top; });
   614→    },
   615→    /* 收集三種任務的候選目標，依「離這個游標多近」排序 */
   616→    options: function (a) {

   623→          var nx = Math.max(r.left, Math.min(a.x, r.right)), ny = Math.max(r.top, Math.min(a.y, r.bottom));
   624→          out.push({ type: pair[0], e: e, d: Math.hypot(nx - a.x, ny - a.y) * rand(.85, 1.15) });
   625→        });
   626→      });
   627→      return out.sort(function (p, q) { return p.d - q.d; });
   628→    },
   629→    /* ---- 讀一行字：慢慢掃過去，留下反白 ---- */
   630→    doRead: function (a, e, L) {
   631→      var self = this, fx = this.fx, h = L.bottom - L.top, y = L.top + h * .5;

   636→        fx.appendChild(hl);

   639→          setTimeout(function () { hl.classList.add('gone'); setTimeout(function () { hl.remove(); }, 900); }, 550);

   641→          a.cont = more ? { e: e, top: L.top } : null;

   644→      });

   646→    /* ---- 點一下按鈕或卡片 ---- */

   673→    /* ---- 框選一張圖或標題，標出尺寸 ---- */

   692→    startTask: function (a) {
   693→      /* 1. 剛讀完的段落還有下一行：優先接著讀 */
   694→      var c = a.cont; a.cont = null;
   695→      if (c && document.contains(c.e) && !this.isBusy(c.e)) {
   696→        var ls = this.lines(c.e);
   697→        for (var i = 0; i < ls.length; i++) if (ls[i].top > c.top + 4) { this.doRead(a, c.e, ls[i]); return true; }
   698→      }
   699→      /* 2. 否則挑附近的東西：越近越優先，保留一點隨機 */
   700→      var opts = this.options(a);
   701→      for (var k = 0; k < opts.length && k < 10; k++) {
   702→        if (k < opts.length - 1 && Math.random() < .3) continue;
   703→        var o = opts[k];
   704→        if (o.type === 'read') {
   705→          var lines = this.lines(o.e);
   706→          if (!lines.length) continue;
   707→          lines.sort(function (p, q) { return Math.abs((p.top + p.bottom) / 2 - a.y) - Math.abs((q.top + q.bottom) / 2 - a.y); });
   708→          this.doRead(a, o.e, lines[0]); return true;
   709→        }
   710→        if (o.type === 'click' && this.doClick(a, o.e)) return true;
   711→        if (o.type === 'inspect' && this.doInspect(a, o.e)) return true;
   712→      }
   713→      return false;
   714→    },
   715→    /* 待機：在做完事的地方小幅度、慢慢地飄 */
   716→    drift: function (a, now, dt) {
   717→      if (!a.anchor) a.anchor = { x: Math.max(30, Math.min(innerWidth - 60, a.x)), y: Math.max(90, Math.min(innerHeight - 50, a.y)) };
   718→      var hx = a.anchor.x + Math.cos(now / 1500 * a.w + a.ph) * 34;
   719→      var hy = a.anchor.y + Math.sin(now / 1900 * a.w + a.ph * 1.3) * 24;
   720→      a.vx += (hx - a.x) * .0012 * dt; a.vy += (hy - a.y) * .0012 * dt;
   721→      a.vx *= .88; a.vy *= .88;
   722→      a.x += a.vx; a.y += a.vy;
   723→    },
   724→    tick: function (now) {
   725→      var dt = Math.min(40, now - (this.last || now)); this.last = now;
   726→      this.leaving.forEach(function (a) { a.stepSeg(dt); a.draw(); });
   727→      var mouseActive = fine && this.mx >= 0 && now - this.lastMove < 2800;
   728→      var running = 0, n = this.agents.length, i, a;
   729→      for (i = 0; i < n; i++) if (this.agents[i].task) running++;
   730→      for (i = 0; i < n; i++) {
   731→        a = this.agents[i];
   732→        if (a.task) { a.stepSeg(dt); a.draw(); continue; }
   733→        if (T.tasks && now >= a.next) {
   734→          if (running < T.maxBusy && this.startTask(a)) { running++; a.stepSeg(dt); a.draw(); continue; }
   735→          a.next = now + rand(2000, 6000);
   736→        }
   737→        if (mouseActive) {
   738→          /* 滑鼠在動：圍著滑鼠跑 */
   739→          if (a.seg || a.queue.length) a.clear();
   740→          a.anchor = null;
   741→          var ang = a.ph + now / 1000 * .55 * a.w;
   742→          var tx = this.mx + Math.cos(ang) * a.r + 10, ty = this.my + Math.sin(ang) * a.r * .8 + 14;
   743→          a.vx += (tx - a.x) * .0016 * dt; a.vy += (ty - a.y) * .0016 * dt;
   744→          for (var j = 0; j < n; j++) {
   745→            if (j === i) continue;
   746→            var b = this.agents[j], dx = a.x - b.x, dy = a.y - b.y, d2 = dx * dx + dy * dy;
   747→            if (d2 < 900 && d2 > .01) { var f = (900 - d2) / 900 * .06 * dt, d = Math.sqrt(d2); a.vx += dx / d * f; a.vy += dy / d * f; }
   748→          }
   749→          a.vx *= .9; a.vy *= .9;
   750→          var sp = Math.hypot(a.vx, a.vy);
   751→          if (sp > T.followMax) { a.vx *= T.followMax / sp; a.vy *= T.followMax / sp; }
   752→          a.x += a.vx; a.y += a.vy;
   753→        } else {
   754→          /* 滑鼠停著：在原地慢慢飄 */
   755→          if (a.seg || a.queue.length) a.stepSeg(dt);
   756→          else this.drift(a, now, dt);
   757→        }
   758→        a.draw();
   759→      }
   760→    }
   761→  };
